Throws InvalidAgentDefinitionError for an invalid name, duplicate trigger ids, or a toolsContext entry naming a chamberu tool or no tool of a plain tool set. See Agents.

ts
/*
 * Copyright (c) 2026.
 * Author Peter Placzek (tada5hi)
 * For the full copyright and license information,
 * view the LICENSE file that was distributed with this source code.
 */

import type { LanguageModel, OutputInterface, ToolSet } from 'ai';
import type { Budget, RunContext } from '../run/types.ts';
import type { StateSchema } from '../state/types.ts';
import type { ToolContext } from '../tool/types.ts';
import type { TriggerDefinition } from '../trigger/types.ts';

/** The factory's argument: the same ToolContext the run's chamberu tools get. */
export type ToolFactoryContext<S extends StateSchema = StateSchema> = ToolContext<S>;

export type ApprovalPolicy = 'user' | ((input: unknown, ctx: { runtimeContext: RunContext }) => 'user' | 'allow' | 'deny');

/**
 * Decides per tool name whether a run may use a tool. A refused tool is not
 * offered to the model (it is left out of `activeTools`). A call to it that
 * still arrives is a tool error, never a run failure, and its tool step
 * records `tool_not_allowed` (ToolNotAllowedError) on both paths:
 * - the model calls it anyway: the AI SDK answers the model with its
 *   unavailable-tool error (NoSuchToolError), the engine records the step
 *   as ToolNotAllowedError;
 * - an approved call from an older history (the policy changed before the
 *   resume) reaches the engine's per-call re-check, which answers the model
 *   with ToolNotAllowedError.
 * Should depend only on the run snapshot (e.g. `run.trigger.invoker`), so
 * the tool set is stable across executions.
 */
export type ToolPolicy = (toolName: string, ctx: { run: RunContext }) => boolean;

export type AgentDefinition<S extends StateSchema = StateSchema> = {
    name: string;
    model: LanguageModel;
    instructions: string;
    triggers?: TriggerDefinition[];
    /**
     * The agent's tools: a plain `ToolSet` (the default), or a factory for
     * dynamic sets, called once per execution with the same ToolContext the
     * tools get. Tools built with `defineTool`/`defineEffectTool` receive that
     * context as `context` in `execute`.
     */
    tools?: ToolSet | ((ctx: ToolContext<S>) => ToolSet);
    /**
     * Per tool name, the context of a tool that is not chamberu's (e.g. a
     * third-party tool with its own `contextSchema`), passed through unchanged
     * by reference to every call of every run: treat it as immutable. An entry
     * naming a chamberu tool or no tool is an InvalidAgentDefinitionError (at
     * defineAgent for a plain set, a failed run for a factory).
     */
    toolsContext?: Record<string, unknown>;
    /** Filters the merged tool set (state tools included) per run, e.g. requireToolPermission(). */
    toolPolicy?: ToolPolicy;
    /** Record<collection, StandardSchemaV1>. */
    state?: S;
    budget?: Partial<Budget>;
    /**
     * Tool name to policy (maps to AI SDK toolApproval). A call needing approval
     * pauses the run in `waiting`; one approval per pause: when a step asks for
     * several, the run waits on the first and the other calls are dropped from
     * the history, so the model requests them again after the resume.
     */
    approvals?: Record<string, ApprovalPolicy>;
    /** Default 24h; on timeout the call is denied. */
    approvalTimeoutMs?: number;
    /** Notifier name registered on the engine. */
    notify?: string;
    /**
     * Structured output (AI SDK `Output`, e.g. `Output.object({ schema })`):
     * a completed run's `output` is the parsed value instead of the text; a
     * final answer that does not parse fails the run with a ValidationError.
     */
    output?: OutputInterface;
};

export type Agent<S extends StateSchema = StateSchema> = {
    readonly name: string;
    readonly definition: Required<Pick<AgentDefinition<S>, 'budget'>> & AgentDefinition<S>;
};

TriggerSource, TriggerDefinition and TriggerEnvelope:

ts
/*
 * Copyright (c) 2026.
 * Author Peter Placzek (tada5hi)
 * For the full copyright and license information,
 * view the LICENSE file that was distributed with this source code.
 */

import type { Artifact } from '../artifact/types.ts';

export type TriggerSource = 'cron' | 'http' | 'webhook' | 'github' | 'mcp';

export type TriggerDefinition<P = unknown> = {
    source: TriggerSource;
    /** Unique within an agent, e.g. 'cron:*\/30 * * * *'. */
    id: string;
    /** Source-specific (cron: { expression, timezone?, overlap }). */
    options: Record<string, unknown>;
    // method signatures (bivariant): a TriggerDefinition<P> fits AgentDefinition.triggers
    concurrencyKey?(payload: P): string | undefined;
    idempotencyKey?(payload: P): string | undefined;
};

/**
 * Who started a run (or answered an approval), as an authenticator resolved
 * it: a snapshot of identity and grants taken at dispatch, never a token.
 * Revoking a grant later does not reach a run that already holds it.
 */
export type Invoker = {
    id: string;
    /**
     * The authenticator's label: `@chamberu/http`'s staticTokens() sets
     * 'token' and authup() sets 'authup'; oidc() keeps the kind its
     * `principal` mapper returns (the docs use 'oidc'); a custom authenticator
     * or a direct dispatch may use its own (e.g. 'user').
     */
    kind: string;
    name?: string;
    permissions: readonly string[];
};

export type TriggerEnvelope<P = unknown> = {
    id: string;
    agent: string;
    trigger: string;
    source: TriggerSource;
    payload: P;
    idempotencyKey?: string;
    concurrencyKey?: string;
    thread?: string;
    /** Files the run was started with, stored as artifacts of the run; the model gets them as file parts. */
    attachments?: Artifact[];
    /**
     * Who started the run; only ever set from an authenticator (never from a
     * payload). Absent for cron and webhook runs. With an invoker the
     * idempotency key is scoped to it, so only the same invoker's dispatch
     * is deduplicated (and gets its own first run).
     */
    invoker?: Invoker;
    receivedAt: Date;
};

RunEngine ​

ts
class RunEngine {
    constructor(opts: RunEngineOptions);

    dispatch(envelope: DispatchEnvelope): Promise<{ runId: string; deduplicated: boolean }>;
    getRun(id: string): Promise<Run | undefined>;
    on(event: RunEvent['type'], listener: (e: RunEvent) => void): () => void;

    start(): Promise<void>;
    stop(): void;
    drain(opts?: { timeoutMs?: number }): Promise<void>;
    idle(): Promise<void>;
    readonly isDraining: boolean;

    cancel(runId: string): Promise<void>;
    resume(token: string, decision: ResumeDecision): Promise<void>;
    tick(): Promise<void>;
    recover(): Promise<number>;
}
MethodDescriptionGuide
dispatchStores a queued run (and its attachments as artifacts); deduplicates by idempotencyKeyTriggers
getRunReads a run from the storeRun Lifecycle
onSubscribes to run.updated, step.appended or run.delta; returns an unsubscribe functionRun Lifecycle
startRecovers, starts polling and starts queued runsRecovery
stopStops polling onlyRecovery
drainWaits for in-flight runs (default 30 s), then marks the rest interruptedRecovery
idleResolves when nothing runs and nothing more can start
cancelCancels locally, in the store, or requests it from the owning replicaRecovery
resumeAnswers the approval a waiting run waits onApprovals
tickDenies expired approvals and re-sends undelivered notifications; runs every pollApprovals
recoverRequeues abandoned runs; runs on start and every pollRecovery

/*
 * Copyright (c) 2026.
 * Author Peter Placzek (tada5hi)
 * For the full copyright and license information,
 * view the LICENSE file that was distributed with this source code.
 */

import type { Agent } from '../agent/types.ts';
import type { ArtifactInput } from '../artifact/types.ts';
import type { IClock } from '../clock/index.ts';
import type { INotifier } from '../notify/types.ts';
import type { Run, Step } from '../run/types.ts';
import type { IStore } from '../store/types.ts';
import type { Invoker, TriggerEnvelope } from '../trigger/types.ts';

/**
 * A piece of model output while it streams; never stored. `step` is the
 * checkpoint step index the step it belongs to is saved under.
 */
export type RunDelta = {
    runId: string;
    step: number;
    kind: 'text' | 'reasoning';
    text: string;
};

export type RunEvent = { type: 'run.updated'; run: Run } |
    { type: 'step.appended'; step: Step } |
    { type: 'run.delta'; delta: RunDelta };

export type RunEngineOptions = {
    store: IStore;
    agents: Agent[];
    notifiers?: Record<string, INotifier>;
    /** Default `${hostname}-${pid}`. */
    owner?: string;
    /** Default 4. */
    maxConcurrentRuns?: number;
    /** Default 60_000, renewed every ttl/3 while running. */
    claimTtlMs?: number;
    /** HMAC secret for resume tokens; must not be empty. */
    resumeSecret: string;
    clock?: IClock;
    /** Receives errors that have no caller to throw to. Default console.error. */
    onError?: (error: unknown, run?: Run) => void;
};

export type DispatchEnvelope = Omit<TriggerEnvelope, 'id' | 'receivedAt' | 'attachments'> &
    Partial<Pick<TriggerEnvelope, 'id' | 'receivedAt'>> & {
        /** Files to start the run with; stored as artifacts of the run before it is created. */
        attachments?: ArtifactInput[];
    };

export type ResumeDecision = {
    approved: boolean;
    reason?: string;
    /** Who answered; recorded on the wait step (\`output.approver\`). */
    approver?: Invoker;
};

Artifacts ​

Artifact (metadata), ArtifactInput (a file to store) and IRunArtifacts (the tool context's artifacts):

ts
/*
 * Copyright (c) 2026.
 * Author Peter Placzek (tada5hi)
 * For the full copyright and license information,
 * view the LICENSE file that was distributed with this source code.
 */

/** Metadata of a file stored for a run (an attachment or a tool output); the bytes are read separately. */
export type Artifact = {
    id: string;
    runId: string;
    agent: string;
    mediaType: string;
    filename?: string;
    /** Byte length of the data. */
    size: number;
    createdAt: Date;
};

/** A file to store: an attachment at dispatch or a tool's `artifacts.put()`. */
export type ArtifactInput = {
    /** `type/subtype`, e.g. `image/png`. */
    mediaType: string;
    filename?: string;
    data: Uint8Array;
};

/**
 * The artifacts of one run, part of the tool context. Not an effect by
 * itself: put `put()` inside a `defineEffectTool` to store a file exactly once.
 */
export interface IRunArtifacts {
    /** Stores a file as an artifact of the run. Throws ValidationError for an invalid input. */
    put(input: ArtifactInput): Promise<Artifact>;
    /** An artifact of this run with its bytes; undefined for an unknown id or another run's artifact. */
    get(id: string): Promise<{ artifact: Artifact; data: Uint8Array } | undefined>;
    /** The run's artifacts, by createdAt then id. */
    list(): Promise<Artifact[]>;
}

isArtifactMediaType(value: unknown): value is string is the media type rule dispatch and artifacts.put() apply: type/subtype (RFC 6838 restricted names) without parameters. An intake layer can check attachments with it before dispatching (@chamberu/http does).

Tools ​

ts
function defineTool<I, O, S extends StateSchema = StateSchema>(
    def: Tool<I, O, ToolContext<S>> & {
        execute: ToolExecuteFunction<I, O, ToolContext<S>>;
        contextSchema?: never;
    },
): Tool<I, O, ToolContext<S>>;

const toolContextSchema: StandardSchemaV1<ToolContext, ToolContext>; // vendor 'chamberu'

defineTool() is tool() with contextSchema: toolContextSchema; the engine hands the run's ToolContext to every tool carrying that schema (by reference, so a copy of the tool keeps it). Its validate returns the context unchanged and rejects a missing one, which fails the run (validation_error) with Type validation failed for tool context (<tool>): No chamberu tool context: the tool was built by another copy of @chamberu/core, or runs outside a chamberu agent. (the marker is the schema object itself, so a tool from a second copy of @chamberu/core gets no context).
